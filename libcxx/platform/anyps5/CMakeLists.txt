cmake_minimum_required(VERSION 3.22)
project(AnyLibcxx LANGUAGES C CXX ASM)
get_filename_component(ANYLIBCXX_LLVM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LLVM_ENABLE_RUNTIMES libcxx CACHE STRING "" FORCE)
set(LLVM_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LIBCXX_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(LIBCXX_INCLUDE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_THREADS OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_FILESYSTEM OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_LOCALIZATION ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_RANDOM_DEVICE OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_MONOTONIC_CLOCK OFF CACHE BOOL "" FORCE)
set(LIBCXX_CXX_ABI system-libcxxabi CACHE STRING "" FORCE)
set(LIBCXX_CXX_ABI_INCLUDE_PATHS "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/include" CACHE STRING "" FORCE)
set(LIBCXX_EXTRA_SITE_DEFINES "_LIBCPP_ABI_PS5;_LIBCPP_PROVIDES_DEFAULT_RUNE_TABLE" CACHE STRING "" FORCE)
set(LIBCXX_ABI_NAMESPACE __adk1 CACHE STRING "" FORCE)
set(LIBCXX_HERMETIC_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_NEW_DELETE_DEFINITIONS ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_ABI_LINKER_SCRIPT OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_PER_TARGET_RUNTIME_DIR OFF CACHE BOOL "" FORCE)
if(WIN32)
    add_compile_definitions(_CRTIMP= __USE_MINGW_ANSI_STDIO=0 __MSVCRT_VERSION__=0x700 _CRT_SECURE_CPP_OVERLOAD_SECURE_NAMES=0)
    include_directories(BEFORE SYSTEM "${CMAKE_CURRENT_SOURCE_DIR}/include/Windows")
endif()
add_subdirectory("${ANYLIBCXX_LLVM_ROOT}/runtimes" "${CMAKE_CURRENT_BINARY_DIR}/3rdparty/llvm-project/runtimes")
configure_file(include/Locale.hpp "${CMAKE_CURRENT_BINARY_DIR}/3rdparty/llvm-project/runtimes/include/c++/v1/Locale.hpp" COPYONLY)
install(FILES include/Locale.hpp DESTINATION include/c++/v1 COMPONENT cxx-headers)
foreach(header cxxabi.h __cxxabi_config.h)
    configure_file("${ANYLIBCXX_LLVM_ROOT}/libcxxabi/include/${header}" "${CMAKE_CURRENT_BINARY_DIR}/3rdparty/llvm-project/runtimes/include/c++/v1/${header}" COPYONLY)
    install(FILES "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/include/${header}" DESTINATION include/c++/v1 COMPONENT cxx-headers)
endforeach()
set_target_properties(cxx_static PROPERTIES OUTPUT_NAME anycxx)
get_target_property(runtimeSources cxx_static SOURCES)
list(FILTER runtimeSources EXCLUDE REGEX "^exception[.]cpp$")
if(WIN32)
    list(FILTER runtimeSources EXCLUDE REGEX "support/win32/")
endif()
set_property(TARGET cxx_static PROPERTY SOURCES "${runtimeSources}")
if(NOT WIN32)
    target_sources(cxx_static PRIVATE src/Scanning.cpp)
endif()
target_sources(cxx_static PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src/Exceptions.cpp"
    "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/stdlib_exception.cpp"
    "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/stdlib_stdexcept.cpp"
    "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/stdlib_typeinfo.cpp")
target_compile_options(cxx_static PRIVATE -fshort-wchar)
target_compile_definitions(cxx_static PRIVATE __NO_CTYPE __errno_location=_Geterrno)
target_include_directories(cxx_static PRIVATE "${ANYLIBCXX_LLVM_ROOT}/libc")

target_include_directories(cxx_static PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(cxx-headers INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_library(anylibcxx_runtime INTERFACE)
add_library(AnyLibcxx::Runtime ALIAS anylibcxx_runtime)
target_compile_features(anylibcxx_runtime INTERFACE cxx_std_20)
target_compile_options(anylibcxx_runtime INTERFACE -nostdinc++ -fshort-wchar)
target_compile_definitions(anylibcxx_runtime INTERFACE __NO_CTYPE __errno_location=_Geterrno)
target_link_options(anylibcxx_runtime INTERFACE -nostdlib++)
target_link_libraries(anylibcxx_runtime INTERFACE "$<LINK_ONLY:cxx_static>" cxx-headers)
add_library(anylibcxx_checks OBJECT tests/src/ContainerChecks.cpp)
set_target_properties(anylibcxx_checks PROPERTIES CXX_EXTENSIONS OFF)
target_compile_features(anylibcxx_checks PRIVATE cxx_std_20)
target_compile_options(anylibcxx_checks PRIVATE -nostdinc++ -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden)
target_include_directories(anylibcxx_checks PRIVATE tests/include)
target_link_libraries(anylibcxx_checks PRIVATE AnyLibcxx::Runtime)

if(WIN32)
    target_sources(cxx_static PRIVATE "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/private_typeinfo.cpp" "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/abort_message.cpp" src/Windows/Math.cpp src/Windows/Strings.cpp src/Windows/Wide.cpp src/Windows/Memory.cpp src/Windows/Stdio.cpp src/Windows/Conversion.cpp src/Windows/ExceptionTransport.cpp src/Windows/Formatting.cpp)
    set_source_files_properties(src/Windows/Math.cpp src/Windows/Strings.cpp src/Windows/Wide.cpp src/Windows/Memory.cpp src/Windows/Stdio.cpp src/Windows/Conversion.cpp src/Windows/ExceptionTransport.cpp src/Windows/Formatting.cpp PROPERTIES COMPILE_OPTIONS "-fno-builtin;-Wall;-Wextra;-Wpedantic;-Werror")
endif()

if(WIN32)
    foreach(header stdio.h stdlib.h wchar.h)
        configure_file("include/Windows/${header}" "${CMAKE_CURRENT_BINARY_DIR}/3rdparty/llvm-project/runtimes/include/anyps5/${header}" COPYONLY)
    endforeach()
    install(DIRECTORY include/Windows/ DESTINATION include/anyps5 COMPONENT cxx-headers)
    target_compile_options(cxx_static PRIVATE -ffunction-sections -fdata-sections)
    target_include_directories(anylibcxx_runtime SYSTEM BEFORE INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include/Windows")
    target_compile_definitions(anylibcxx_runtime INTERFACE _CRTIMP= __USE_MINGW_ANSI_STDIO=0 __MSVCRT_VERSION__=0x700 _CRT_SECURE_CPP_OVERLOAD_SECURE_NAMES=0)
    set(windowsImports)
    foreach(source Math Strings Wide Memory Stdio Conversion ExceptionTransport Formatting)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/Windows/${source}.cpp")
        file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/Windows/${source}.cpp" sourceText)
        string(REGEX MATCHALL "[A-Za-z_][A-Za-z0-9_]*_nid_postfix" sourceImports "${sourceText}")
        list(APPEND windowsImports ${sourceImports})
    endforeach()
    list(REMOVE_DUPLICATES windowsImports)
    list(REMOVE_ITEM windowsImports __stdinp_nid_postfix __stdoutp_nid_postfix __stderrp_nid_postfix)
    list(APPEND windowsImports "__stdinp_nid_postfix DATA" "__stdoutp_nid_postfix DATA" "__stderrp_nid_postfix DATA")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/WindowsImports.cmake" "set(ANYLIBCXX_WINDOWS_IMPORTS \"${windowsImports}\")\n")
endif()

set(ANYLIBCXX_RUNTIME_RELATIVE "3rdparty/llvm-project/runtimes")
configure_file(cmake/AnyLibcxxConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/AnyLibcxxConfig.cmake" @ONLY)
set(ANYLIBCXX_RUNTIME_RELATIVE "../../..")
configure_file(cmake/AnyLibcxxConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/install/AnyLibcxxConfig.cmake" @ONLY)
install(FILES "$<TARGET_FILE:cxx_static>" DESTINATION lib COMPONENT anylibcxx)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/install/AnyLibcxxConfig.cmake" DESTINATION lib/cmake/AnyLibcxx COMPONENT anylibcxx)
if(WIN32)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/WindowsImports.cmake" DESTINATION lib/cmake/AnyLibcxx COMPONENT anylibcxx)
endif()
install(FILES "${ANYLIBCXX_LLVM_ROOT}/libcxx/LICENSE.TXT" DESTINATION . COMPONENT cxx-headers)
if(NOT ANYLIBCXX_TARGET MATCHES "^(windows|linux|ps5)$")
    message(FATAL_ERROR "Use an AnyLibcxx platform toolchain")
endif()
set(ANYLIBCXX_PACKAGE_VERSION "dev" CACHE STRING "Package version")
if(NOT ANYLIBCXX_PACKAGE_VERSION MATCHES "^[A-Za-z0-9][A-Za-z0-9._-]*$")
    message(FATAL_ERROR "Invalid package version")
endif()
set(ANYLIBCXX_PACKAGE_PLATFORM "${ANYLIBCXX_TARGET}")
configure_file(cmake/Package.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/Package.cmake" @ONLY)
add_custom_target(anylibcxx_headers
    COMMAND "${CMAKE_COMMAND}" -DHEADERS_ONLY=ON -P "${CMAKE_CURRENT_BINARY_DIR}/Package.cmake"
    DEPENDS generate-cxx-headers
    VERBATIM)
add_custom_target(anylibcxx_package
    COMMAND "${CMAKE_COMMAND}" -DHEADERS_ONLY=OFF -P "${CMAKE_CURRENT_BINARY_DIR}/Package.cmake"
    DEPENDS cxx_static anylibcxx_checks
    VERBATIM)

if(WIN32)
    enable_testing()
    add_executable(anylibcxx_math_abi tests/src/MathAbi.cpp tests/src/ClobberVolatileRegisters.S src/Windows/Math.cpp)
    target_compile_features(anylibcxx_math_abi PRIVATE cxx_std_20)
    target_compile_options(anylibcxx_math_abi PRIVATE -fno-builtin -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME math_abi COMMAND anylibcxx_math_abi)
endif()
