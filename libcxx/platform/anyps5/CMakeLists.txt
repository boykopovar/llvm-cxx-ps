cmake_minimum_required(VERSION 3.22)
project(AnyLibcxx LANGUAGES C CXX ASM)
get_filename_component(ANYLIBCXX_LLVM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LLVM_ENABLE_RUNTIMES libcxx CACHE STRING "" FORCE)
set(LLVM_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(LLVM_INCLUDE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LIBCXX_INCLUDE_TESTS OFF CACHE BOOL "" FORCE)
set(LIBCXX_INCLUDE_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_THREADS OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_FILESYSTEM OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_LOCALIZATION ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_RANDOM_DEVICE OFF CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_MONOTONIC_CLOCK OFF CACHE BOOL "" FORCE)
set(LIBCXX_CXX_ABI system-libcxxabi CACHE STRING "" FORCE)
set(LIBCXX_CXX_ABI_INCLUDE_PATHS "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/include" CACHE STRING "" FORCE)
set(LIBCXX_EXTRA_SITE_DEFINES "_LIBCPP_ABI_PS5;_LIBCPP_PROVIDES_DEFAULT_RUNE_TABLE" CACHE STRING "" FORCE)
set(LIBCXX_ABI_NAMESPACE __adk1 CACHE STRING "" FORCE)
set(LIBCXX_HERMETIC_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_NEW_DELETE_DEFINITIONS ON CACHE BOOL "" FORCE)
set(LIBCXX_ENABLE_ABI_LINKER_SCRIPT OFF CACHE BOOL "" FORCE)
set(LLVM_ENABLE_PER_TARGET_RUNTIME_DIR OFF CACHE BOOL "" FORCE)
add_subdirectory("${ANYLIBCXX_LLVM_ROOT}/runtimes" "${CMAKE_CURRENT_BINARY_DIR}/3rdparty/llvm-project/runtimes")
configure_file(include/Locale.hpp "${CMAKE_CURRENT_BINARY_DIR}/3rdparty/llvm-project/runtimes/include/c++/v1/Locale.hpp" COPYONLY)
install(FILES include/Locale.hpp DESTINATION include/c++/v1)
get_target_property(runtimeSources cxx_static SOURCES)
list(FILTER runtimeSources EXCLUDE REGEX "^exception[.]cpp$")
set_property(TARGET cxx_static PROPERTY SOURCES "${runtimeSources}")
target_sources(cxx_static PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src/Exceptions.cpp"
    "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/stdlib_exception.cpp"
    "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/stdlib_stdexcept.cpp"
    "${ANYLIBCXX_LLVM_ROOT}/libcxxabi/src/stdlib_typeinfo.cpp")
target_compile_options(cxx_static PRIVATE -fshort-wchar)
target_compile_definitions(cxx_static PRIVATE __NO_CTYPE __errno_location=_Geterrno)
target_include_directories(cxx_static PRIVATE "${ANYLIBCXX_LLVM_ROOT}/libc")

target_include_directories(cxx_static PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(cxx-headers INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_library(anylibcxx_runtime INTERFACE)
add_library(AnyLibcxx::Runtime ALIAS anylibcxx_runtime)
target_compile_features(anylibcxx_runtime INTERFACE cxx_std_20)
target_compile_options(anylibcxx_runtime INTERFACE -nostdinc++ -fshort-wchar)
target_compile_definitions(anylibcxx_runtime INTERFACE __NO_CTYPE __errno_location=_Geterrno)
target_link_options(anylibcxx_runtime INTERFACE -nostdlib++)
target_link_libraries(anylibcxx_runtime INTERFACE "$<LINK_ONLY:cxx_static>" cxx-headers)
add_library(anylibcxx_checks OBJECT tests/src/ContainerChecks.cpp)
set_target_properties(anylibcxx_checks PROPERTIES CXX_EXTENSIONS OFF)
target_compile_features(anylibcxx_checks PRIVATE cxx_std_20)
target_compile_options(anylibcxx_checks PRIVATE -nostdinc++ -Wall -Wextra -Wpedantic -Werror -fvisibility=hidden)
target_include_directories(anylibcxx_checks PRIVATE tests/include)
target_link_libraries(anylibcxx_checks PRIVATE AnyLibcxx::Runtime)
