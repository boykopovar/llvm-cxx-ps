cmake_minimum_required(VERSION 3.22.1)
project(AnyLibcxxRuntimeCheck LANGUAGES C CXX)
set(ADK_TARGET "${ANYLIBCXX_TARGET}")
find_package(AnyDevKit CONFIG REQUIRED)
adk_add_executable(RuntimeCheck Main.cpp ../src/ContainerChecks.cpp ../src/MutexChecks.cpp ../src/CApiChecks.cpp ../src/ThreadChecks.cpp)
target_include_directories(RuntimeCheck PRIVATE ../include)
target_link_options(RuntimeCheck PRIVATE "$<$<AND:$<PLATFORM_ID:Windows>,$<CXX_COMPILER_ID:GNU>>:-Wl,--gc-sections>")
set_target_properties(RuntimeCheck PROPERTIES CXX_EXTENSIONS OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime")
adk_link_libraries(RuntimeCheck libc libkernel)

adk_add_executable(NoexceptCheck Noexcept.cpp)
set_target_properties(NoexceptCheck PROPERTIES CXX_EXTENSIONS OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/noexcept")
adk_link_libraries(NoexceptCheck libc libkernel)
enable_testing()
add_test(NAME runtime COMMAND RuntimeCheck)
set_tests_properties(runtime PROPERTIES TIMEOUT 30)
add_test(NAME noexcept COMMAND NoexceptCheck)
set_tests_properties(noexcept PROPERTIES TIMEOUT 10)
