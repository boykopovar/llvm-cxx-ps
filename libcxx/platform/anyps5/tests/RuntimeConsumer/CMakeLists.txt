cmake_minimum_required(VERSION 3.22.1)
project(AnyLibcxxRuntimeCheck LANGUAGES C CXX)
set(ADK_TARGET "${ANYLIBCXX_TARGET}")
find_package(AnyDevKit CONFIG REQUIRED)
adk_add_executable(RuntimeCheck Main.cpp ../src/ContainerChecks.cpp ../src/MutexChecks.cpp ../src/CApiChecks.cpp)
target_include_directories(RuntimeCheck PRIVATE ../include)
set_target_properties(RuntimeCheck PROPERTIES CXX_EXTENSIONS OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/runtime")
adk_link_libraries(RuntimeCheck libc libkernel)
enable_testing()
add_test(NAME runtime COMMAND RuntimeCheck)
set_tests_properties(runtime PROPERTIES TIMEOUT 30)
