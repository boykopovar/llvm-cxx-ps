cmake_minimum_required(VERSION 3.22)
project(AnyLibcxxAllocationChecks LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(allocation_heap ../src/AllocationHeap.cpp ../../src/Allocation/Heap.cpp ../../src/Allocation/Api.cpp)
foreach(target allocation_heap)
    target_include_directories(${target} PRIVATE ../../include)
    target_compile_features(${target} PRIVATE cxx_std_20)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(${target} PRIVATE -fno-builtin -Wall -Wextra -Wpedantic -Werror)
endforeach()
target_link_libraries(allocation_heap PRIVATE Threads::Threads)
enable_testing()
add_test(NAME allocation_heap COMMAND allocation_heap)
